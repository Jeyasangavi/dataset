Relay_server.py


import socket
import threading
import json
import sys


N = 3
MODE = sys.argv[1] if len(sys.argv) > 1 else "ricart"


clients = {}
send_locks = {}
lock = threading.Lock()
ready = False




def send(pid, message):
    with lock:
        conn = clients.get(pid)
        slock = send_locks.get(pid)


    if conn:
        try:
            with slock:
                conn.sendall((json.dumps(message) + "\n").encode())
        except OSError:
            pass




def handle(conn):
    global ready


    reader = conn.makefile("r")
    pid = None


    try:
        registration = json.loads(reader.readline())
        pid = registration["pid"]


        with lock:
            clients[pid] = conn
            send_locks[pid] = threading.Lock()
            print("Connected:", pid)


            if len(clients) == N and not ready:
                ready = True
                all_ready = list(clients.keys())
            else:
                all_ready = []


        for p in all_ready:
            send(p, {"type": "READY"})


        if all_ready and MODE == "token":
            send(1, {"type": "TOKEN"})


        for line in reader:
            message = json.loads(line)
            target = message["to"]
            payload = message["payload"]
            send(target, payload)


    except (OSError, ValueError, KeyError):
        pass
    finally:
        if pid is not None:
            with lock:
                clients.pop(pid, None)
                send_locks.pop(pid, None)
            print("Disconnected:", pid)
        conn.close()




server = socket.socket()
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("127.0.0.1", 5000))
server.listen()


print("Relay server started:", MODE)


while True:
    conn, _ = server.accept()
    threading.Thread(
        target=handle, args=(conn,), daemon=True
    ).start()

import socket
import threading
import json
import sys
import time


pid = int(sys.argv[1])
N = 3


sock = socket.socket()
sock.connect(("127.0.0.1", 5000))
sock.sendall((json.dumps({"pid": pid}) + "\n").encode())


send_lock = threading.Lock()
condition = threading.Condition()
ready = threading.Event()


clock = 0
state = "RELEASED"
request_time = 0
replies = set()
deferred = []




def send(to, payload):
    message = {"to": to, "payload": payload}
    with send_lock:
        sock.sendall((json.dumps(message) + "\n").encode())




def receive_messages():
    global clock, state


    reader = sock.makefile("r")


    for line in reader:
        msg = json.loads(line)
        kind = msg["type"]


        if kind == "READY":
            ready.set()
            continue


        if kind == "REPLY":
            with condition:
                replies.add(msg["from"])
                condition.notify_all()


        elif kind == "REQUEST":
            sender = msg["from"]
            timestamp = msg["ts"]


            with condition:
                clock = max(clock, timestamp) + 1


                defer = (
                    state == "HELD"
                    or (
                        state == "WANTED"
                        and (request_time, pid) < (timestamp, sender)
                    )
                )


                if defer:
                    deferred.append(sender)


            if defer:
                print(f"P{pid} defers reply to P{sender}")
            else:
                send(sender, {"type": "REPLY", "from": pid})




threading.Thread(target=receive_messages, daemon=True).start()


print(f"P{pid} connected; waiting for other processes...")
ready.wait()
print(f"P{pid} ready. Type r to request CS, q to quit.")


while True:
    command = input().strip().lower()


    if command == "q":
        break


    if command != "r":
        continue


    with condition:
        if state != "RELEASED":
            print("Already requesting or using CS")
            continue


        clock += 1
        request_time = clock
        state = "WANTED"
        replies.clear()


    print(f"P{pid} sends REQUEST")
    for other in range(1, N + 1):
        if other != pid:
            send(other, {
                "type": "REQUEST",
                "from": pid,
                "ts": request_time
            })


    with condition:
        while len(replies) < N - 1:
            condition.wait()
        state = "HELD"


    print(f"P{pid} ENTERS CS")
    time.sleep(2)
    print(f"P{pid} EXITS CS")


    with condition:
        state = "RELEASED"
        pending = deferred[:]
        deferred.clear()


    for other in pending:
        send(other, {"type": "REPLY", "from": pid})


sock.close()

Coordinator_server.py


import socket
import threading
import json


clients = {}
queue = []
busy = False
lock = threading.Lock()




def send(conn, message, send_lock):
    with send_lock:
        conn.sendall((json.dumps(message) + "\n").encode())




def grant_next():
    global busy


    if queue and not busy:
        pid = queue[0]
        busy = True
        conn, send_lock = clients[pid]
        print("Coordinator sends GRANT to", pid)
        send(conn, {"type": "GRANT"}, send_lock)




def handle(conn):
    global busy


    reader = conn.makefile("r")
    pid = None
    send_lock = threading.Lock()


    try:
        registration = json.loads(reader.readline())
        pid = registration["pid"]


        with lock:
            clients[pid] = (conn, send_lock)


        print("Process connected:", pid)


        for line in reader:
            msg = json.loads(line)


            with lock:
                if msg["type"] == "REQUEST":
                    print(pid, "sends REQUEST")


                    if pid not in queue:
                        queue.append(pid)


                    grant_next()


                elif msg["type"] == "RELEASE":
                    print(pid, "sends RELEASE")


                    if queue and queue[0] == pid:
                        queue.pop(0)
                        busy = False
                        grant_next()


    except (OSError, ValueError, KeyError):
        pass
    finally:
        with lock:
            clients.pop(pid, None)
            if pid in queue:
                was_first = queue[0] == pid
                queue.remove(pid)


                if was_first:
                    busy = False
                    grant_next()


        conn.close()
        print("Disconnected:", pid)




server = socket.socket()
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("127.0.0.1", 5001))
server.listen()


print("Coordinator server started on port 5001")


while True:
    conn, _ = server.accept()
    threading.Thread(
        target=handle, args=(conn,), daemon=True
    ).start()

class Process:


    def __init__(self, name):
        self.name = name
        self.clock = 0


    def event(self):
        self.clock += 1
        print(self.name, "event:", self.clock)


    def send(self):
        self.clock += 1
        print(self.name, "sends message:", self.clock)
        return self.clock


    def receive(self, timestamp):
        self.clock = max(self.clock, timestamp) + 1
        print(self.name, "receives message:", self.clock)

Vector clock:


class Process:


    def __init__(self, pid, n):
        self.pid = pid
        self.clock = [0] * n


    def event(self):
        self.clock[self.pid] += 1
        print("P", self.pid + 1, "event:", self.clock)


    def send(self):
        self.clock[self.pid] += 1
        print("P", self.pid + 1, "sends:", self.clock)
        return self.clock.copy()


    def receive(self, received):
        for i in range(len(self.clock)):
            self.clock[i] = max(self.clock[i], received[i])


        self.clock[self.pid] += 1


        print("P", self.pid + 1, "receives:", self.clock)




p1 = Process(0, 3)
p2 = Process(1, 3)


p1.event()


message = p1.send()


p2.receive(message)


p2.event()

## 1. Load Balancing and Fault Tolerance

Aim: Distribute requests across multiple servers and maintain service availability if one server fails.

### Part A — Configure the load balancer

Use Server1 and Server2 as web servers and Server3 as the load balancer.

On Server3 — Ubuntu Terminal:

```
sudo apt update
sudo apt install haproxy -y
sudo nano /etc/haproxy/haproxy.cfg
```

Go to the end of the configuration file and add this configuration. Replace the IP addresses with your actual Server1 and Server2 addresses.

```
frontend web_front
    bind *:80
    default_backend web_back

backend web_back
    balance roundrobin
    server server1 192.168.119.129:80 check
    server server2 192.168.119.130:80 check
```

Here, `roundrobin` means requests are distributed between the servers in turn.

Save the file: press `Ctrl + O`, press `Enter`, then `Ctrl + X`.

If the existing HAProxy configuration already has a frontend binding to port 80, modify that existing configuration rather than creating a second conflicting listener.

Validate and start HAProxy:

```
sudo haproxy -c -f /etc/haproxy/haproxy.cfg
sudo systemctl enable --now haproxy
sudo systemctl restart haproxy
sudo systemctl status haproxy
```

Look for `active (running)` and press `q`.

### Part B — Test load balancing

On the computer or VM acting as the client, open a browser and visit:

```
http://SERVER3_IP
```

Alternatively, from an Ubuntu terminal, run:

```
curl http://SERVER3_IP
```

Replace `SERVER3_IP` with Server3's actual address.

Repeat the request several times. The responses should alternate between the Server1 and Server2 pages, provided both are healthy and both have distinct page content.

### Part C — Demonstrate fault tolerance

For a simple manual failover demonstration:

1. Keep Server2 available as the backup.
2. Copy the website files from Server1 to Server2.
3. Test the website directly on Server2.
4. Use VMware Workstation to select Server1 and click Power Off.
5. Open `http://SERVER2_IP` in the browser to demonstrate that the backup server still serves the website.

To synchronize the website files, first install SSH and `rsync` on the VMs. On Server2, run:

```
sudo apt update
sudo apt install openssh-server rsync -y
sudo systemctl enable --now ssh
```

On Server1, run the following, replacing `USER` with Server2's Ubuntu username and `SERVER2_IP` with its actual address:

```
sudo apt install rsync openssh-client -y
sudo rsync -av /var/www/html/ USER@SERVER2_IP:/tmp/backup-html/
```

Then, on Server2, copy the synchronized files into Apache's website folder:

```
sudo cp /tmp/backup-html/index.html /var/www/html/index.html
```

This demonstrates backup and manual recovery. For a real production setup, automatic failover would need additional configuration; powering off Server1 alone does not automatically make Server2 take over the same IP address.

Expected result: Requests are distributed across healthy web servers, and the backup server can serve the website if the primary is unavailable.
